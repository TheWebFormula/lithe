Example: JWT refresh token

In this example we will create a fetcher instance that automatically handle refreshing short lived JWT access tokens.

${`import { Fetcher, Interceptor } from '@thewebformula/lithe';

// Setup refresh token interceptor
const refreshToken = new Interceptor({
  // Only intercept 401 status codes
  statusCodes: [401],

  /**
  * pause all requests until after handler is resolved
  * this will prevent other requests from failing and calling the /token endpoint
  */
  waitForResoltuion: true,

  async after(request, response) {
    // Get access token from refresh token stored in cookie
    const resp = await tokenIssuerFetcher.get('/token');
    const token = await resp.json();

    // set header on api fetcher
    apiFetcher.headers.set('Authorization', \`Bearer \${token}\`);

    // rerun original request
    return request;
  }
});

export const apiFetcher = new Fetcher({
  baseURL: 'http://localhost:3002',
  credentials: 'include',
  interceptors: [refreshToken]
});

// Setup token issuer instance
export const tokenIssuerFetcher = new Fetcher({
  baseURL: 'http://localhost:3001',
  credentials: 'include'
});`}
        

${`import { tokenIssuerFetcher, apiFetcher } from './apiFetcher.js';

async function login() {
  const resp = await tokenIssuerFetcher.post('/login', {
    headers: { Authorization: \`Basic \${btoa('user:pass')}\` }
  });
  if (resp.status !== 201) throw Error('Cannot authenticate');
}

async function getData() {
  const resp = await apiFetcher.get('/token');
  return resp.json();
}`}

Creating instances

You can make instances to support needs like multiple domains, auth vs no auth, and more.

${`import { Fetcher } from '@thewebformula/lithe';

const fetcherInstance = new Fetcher({
  // base url for all requests
  baseURL: 'http://localhost:3000',

  // credentials for all requests
  credentials: 'include',

  // headers for all requests (you can still add more when making requests)
  headers: {
    'api-version': 'v1'
  }
});

// using the instance to make a request
// uses the same interface as fetch
const resp = await fetcherInstance.fetch('/send-data', {
  method: 'POST',

  // JSON will automatically be stringified and the Content-Type header will be set to application/json
  body: {
    name: 'John Doe',
    age: 30
  }
});


// retrive data the same as fetch
const data = await resp.json();

// Convenience methods
await fetcherInstance.get('/get-data', {...});
await fetcherInstance.post('/send-data', {...});
await fetcherInstance.put('/send-data', {...});
await fetcherInstance.patch('/send-data', {...});
await fetcherInstance.delete('/send-data', {...});
await fetcherInstance.head('/send-data', {...});

// Update headers on instance
fetcherInstance.headers.set('Authorization', \`Bearer \${token}\`);`}

Fetcher interceptors

Interceptors allow you to handle items like authentication refresh, pre-processing data, header manipulation and more.

${`import { Fetcher, Interceptor } from '@thewebformula/lithe';

// Setup refresh token interceptor
const setUserHeader = new Interceptor({
  // Runs before fetch passing in the Request object https://developer.mozilla.org/en-US/docs/Web/API/Request
  async before(request) {
    // set headers
    request.headers.set('user-id', userId);
    return request; // will use passed in request if none returned

    // You can build a new Request object and return that also
    // return new Request({ headers: { 'user-id': userId } });
  }
});


// Error handler
const setUserHeader = new Interceptor({
  // Only intercept listed status codes
  // If not set then all requests are intercepted
  statusCodes: [500, 403],

  // pause all requests until after handler is resolved
  // includes the request run from the return
  // this will prevent other requests from failing and calling the /token endpoint
  waitForResoltuion: false,

  async after(request, response) {
    if (response.status === 403) {
      // do something
    } else if (response.status === 500) {
      // do something else
    }

    // if you return nothing that will be the end of the request

    // if you return a Request object then that will be run
    // this allowes you to handle things like refreshing expired tokens
    // return new Request();
  }
});

const fetcherInstance = new Fetcher({
  baseURL: 'http://localhost:3000',
  interceptors: [setUserHeader, errorHandler],
});`}
