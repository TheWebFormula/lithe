${`import { Component, html, Cue } from '@thewebformula/lithe';

class TemplateConditionalPage extends Component {
  static title = 'Templating conditionals';

  shouldShow = new Cue.State(true);

  toggleShow() {
    this.shouldShow.set(!this.shouldShow.get());
  }

  template() {
    return html\`
      // wrapping the expression in html() will act like a compute
      \${html(() => this.shouldShow.get() ? 'Showing' : 'hiding')}
    \`;
  }
}
customElements.define('template-lists-page', TemplateConditionalPage);`}

${`import { Component, html, Cue } from '@thewebformula/lithe';

class TemplateConditionalPage extends Component {
  static title = 'Templating conditionals';

  switchOneTwo = new Cue.State(true);

  switch() {
    this.switchOneTwo.set(!this.switchOneTwo.get());
  }

  template() {
    return html\`
      // wrapping the expression in html() will act like a compute
      \$html(() => this.switchOneTwo.get()
        ? html\` this.switch()}>Switch: one\`
        : html\` this.switch()}>Switch: two\`
      )}
    \`;
  }
}
customElements.define('template-lists-page', TemplateConditionalPage);`}
