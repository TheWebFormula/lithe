You can use the Component class to build webcomponent. This extends the default capabilities of web components.

  Load data

${`import { Component, html, css } from '@thewebformula/lithe';

class MyButton extends Component {
  static useShadowRoot = true;
  static shadowRootDelegateFocus = true;
  static styles = css\`
    button:not(.loading) .loading-text { display: none; }
    button.loading .text { display: none; }
  \`;

  static observedAttributesExtended = {
    disabled: { type: 'toggle' },
    loading: { type: 'toggle' }
  };

  attributeChangedCallbackExtended(name, oldValue, newValue) {
    this[name] = newValue;
  }

  #button;
  #disabled = false;
  get disabled() { return this.#disabled; }
  set disabled(value) {
    this.#disabled = value;
    this.#button.disabled = value;
  }

  #loading = false;
  get loading() { return this.#loading; }
  set loading(value) {
    this.#loading = value;
    this.#button.classList.toggle('loading', value);
  }

  afterRender() {
    this.#button = this.shadowRoot.querySelector('button');
  }

  handleClick(event) {
    if (this.#loading) {
      event.stopImmediatePropagation();
      event.preventDefault();
    }
  }

  template() {
    return html\`
      
    \`;
  }
}
customElements.define('my-button', MyButton);`}

Component breakdown

${`import { Component } from '@thewebformula/lithe';

/** You can import html
* html imports as a string
*/
import htmlTemplate from './component.html';

/** You can import stylesheets
* Stylesheets imports as modules that export CSSStyleSheet
* You can add them to the component 'static styles = [styles];'
*/
import styles from '../styles.css' assert { type: 'css' };



class CustomButton extends Component {
  /**
    * Pass in HTML string. Use for imported .HTML
    *   Supports template literals: 
\${this.var}

    *   alternative to using the template() method
    * @type {String}
    */
  static htmlTemplate = htmlTemplate;


  /**
    * Hook up shadow root
    * @type {Boolean}
    * @default false
    */
  static useShadowRoot = false;


  /**
    * @type {Boolean}
    * @default false
    */
  static shadowRootDelegateFocus = false;


  /**
    * Pass in styles.
    *   StyleSheetTemplate: Can use css tag: css\`h1 { color: blue; }\`
    *   CSSStyleSheet: Can use imported stylesheets: import styles from '../styles.css' assert { type: 'css' };
    * @type {CSSStyleSheet | StyleSheetTemplate | (CSSStyleSheet | StyleSheetTemplate)[]}
    */
  static styles;


  /**
    * @typedef {String} AttributeType
    * @value 'string' Convert to a string. null = ''
    * @value 'number' Convert to a number. isNaN = ''
    * @value 'int' Convert to a int. isNaN = ''
    * @value 'toggle' Add / remove attribute
    * @value 'boolean' Convert to a boolean
    * @value 'object' Pass in an objecty. This will add teh attribute with no value
    * @value 'event' Allows code to be executed. Similar to onchange="console.log('test')"
    */
  /**
    * @typedef {Object} AttributeConfig
    * @property {AttributeType} type - The parsed primitive type of the attribute
    * @property {Boolean} [reflect=true] - Whether the attribute should be reflected to the DOM (converted to dash case when reflected)
    */
  /**
    * Extended observedAttributes, allowing you to specify types
    * @static
    * @type {Record}
    */
  static observedAttributesExtended = { require: { type: 'toggle' } };


  /**
    * Use with observedAttributesExtended
    * Attribut names are automatically converted from dashcase to camel case
    * @function
    * @param {String} name - Attribute name (converted to camel case)
    * @param {String} oldValue - Old attribute value
    * @param {String} newValue - New attribute value
    */
  attributeChangedCallbackExtended(name, oldValue, newValue) { }


  /**
    * Method that returns a html template string. This is an alternative to use static htmlTemplate
    * @name template
    * @function
    * @return {TemplateInstance}
    */
  template() { return html'
' }

  connectedCallback() {
    super.connectedCallback(); // must be called if using connectedCallback
  }

  // Rendering only happens once unless called manually
  beforeRender() {}
  afterRender() {
    this.#root.querySelector('button').addEventListener('click', this.#onClick_bound);
  }

  disconnectedCallback() {
    super.disconnectedCallback(); // must be called if using disconnectedCallback
    this.#root.querySelector('button').removeEventListener('click', this.#onClick_bound);
  }

  #onClick() {
    console.log('Custom button component clicked!');
  }
  #onClick_bound = this.#onClick.bind(this);
}

// define web component
customElements.define('custom-button', CustomButton);`}
